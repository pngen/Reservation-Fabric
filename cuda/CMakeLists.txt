# CUDA proof: built only when a CUDA toolkit + sm_120-capable device are available.
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  add_executable(rf_cuda_proof rf_cuda_proof.cu)
  target_include_directories(rf_cuda_proof PRIVATE ${CMAKE_SOURCE_DIR}/include)
  target_link_libraries(rf_cuda_proof PRIVATE ReservationFabric::ReservationFabric)
  set_target_properties(rf_cuda_proof PROPERTIES CUDA_ARCHITECTURES 120 CUDA_STANDARD 20)
  if(MSVC)
    set_property(TARGET rf_cuda_proof PROPERTY CUDA_FLAGS "-ccbin cl -MD -Xcompiler /MD")
  endif()
  add_test(NAME rf_cuda_proof COMMAND $<TARGET_FILE:rf_cuda_proof>)
endif()
